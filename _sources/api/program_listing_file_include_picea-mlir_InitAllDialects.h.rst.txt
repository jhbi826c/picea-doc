   inline void registerAllPICEAMLIRDialects(DialectRegistry &registry)
   {
       // Base2
       registry.insert<base2::Base2Dialect>();
       base2::registerBufferizableOpInterfaceExternalModels(registry);
       // CGRA
       registry.insert<cgra::CGRADialect>();
       // DFG
       registry.insert<dfg::DFGDialect>();
       dfg::registerBufferizableOpInterfaceExternalModels(registry);
       // EmitHLS
       registry.insert<emithls::EmitHLSDialect>();
       // Filter
       registry.insert<filter::FilterDialect>();
   }
   
   inline void registerAllPICEAMLIRDialects(MLIRContext &context)
   {
       DialectRegistry registry;
       registerAllPICEAMLIRDialects(registry);
       context.appendDialectRegistry(registry);
   }
