   void FIROp::getAsmResultNames(function_ref<void(Value, StringRef)> setNameFn)
   {
       Block* block = getOperation()->getBlock();
       int count = 0;
   
       for (auto &opi : *block) {
           if (&opi == getOperation()) break;
           if (opi.getName() == getOperation()->getName()) ++count;
       }
   
       setNameFn(getResult(), "fir_" + std::to_string(count));
   }

   ParseResult FIROp::parse(OpAsmParser &parser, OperationState &state)
   {
       OpAsmParser::UnresolvedOperand input, coefficient;
       Type inputType, coefficientType;
   
       if (parser.parseOperand(input) || parser.parseKeyword("coeff")
           || parser.parseEqual() || parser.parseOperand(coefficient))
           return failure();
   
       if (parser.parseColon() || parser.parseType(inputType) || parser.parseLess()
           || parser.parseArrow() || parser.parseType(coefficientType))
           return failure();
   
       if (parser.resolveOperands(
               {input, coefficient},
               {inputType, coefficientType},
               parser.getCurrentLocation(),
               state.operands))
           return failure();
       state.addTypes(inputType);
   
       return success();
   }

   ParseResult SOSOp::parse(OpAsmParser &parser, OperationState &state)
   {
       OpAsmParser::UnresolvedOperand input, coefficient, scale;
       Type inputType, coefficientType, scaleType;
       bool hasScale = false;
   
       if (parser.parseOperand(input) || parser.parseKeyword("coeff")
           || parser.parseEqual() || parser.parseOperand(coefficient))
           return failure();
       if (succeeded(parser.parseOptionalKeyword("scale"))) {
           hasScale = true;
           if (parser.parseEqual() || parser.parseOperand(scale)) return failure();
       }
   
       if (parser.parseColon() || parser.parseType(inputType) || parser.parseLess()
           || parser.parseArrow() || parser.parseType(coefficientType))
           return failure();
       if (hasScale) {
           if (parser.parseComma() || parser.parseType(scaleType))
               return failure();
       }
   
       if (parser.resolveOperands(
               {input, coefficient},
               {inputType, coefficientType},
               parser.getCurrentLocation(),
               state.operands))
           return failure();
       if (hasScale) {
           if (parser.resolveOperand(scale, scaleType, state.operands))
               return failure();
       }
       state.addTypes(inputType);
   
       return success();
   }

   LogicalResult SOSOp::verify()
   {
       auto coefficient = getCoefficient();
       auto coefficientType = coefficient.getType();
       if (coefficientType.getDimSize(1) != 5)
           return emitOpError("expect [b0, b1, b2, a1, a2] for each section");
   
       auto scale = getScale();
       if (scale) {
           auto scaleType = scale.getType();
           if (scaleType.getDimSize(0) != coefficientType.getDimSize(0) + 1)
               return emitOpError("expect \"section+1\" number of scales");
       }
   
       return success();
   }

   ParseResult EnvelopeOp::parse(OpAsmParser &parser, OperationState &state)
   {
       OpAsmParser::UnresolvedOperand input;
       int32_t windowSize, step;
       StringRef styleStr;
       Type inputType, filteredType;
   
       if (parser.parseOperand(input) || parser.parseLBrace()
           || parser.parseKeyword("window_size") || parser.parseEqual()
           || parser.parseInteger(windowSize) || parser.parseComma()
           || parser.parseKeyword("step") || parser.parseEqual()
           || parser.parseInteger(step) || parser.parseRBrace()
           || parser.parseKeyword("style") || parser.parseEqual()
           || parser.parseKeyword(&styleStr))
           return failure();
   
       auto style = symbolizeEnvelopeStyle(styleStr);
       if (!style.has_value())
           return parser.emitError(parser.getNameLoc(), "unknown envelope style");
   
       if (parser.parseColon() || parser.parseType(inputType)
           || parser.parseArrow() || parser.parseType(filteredType))
           return failure();
   
       if (parser.resolveOperand(input, inputType, state.operands))
           return failure();
       state.addAttribute(
           getWindowSizeAttrName(state.name),
           parser.getBuilder().getI32IntegerAttr(windowSize));
       state.addAttribute(
           getStepAttrName(state.name),
           parser.getBuilder().getI32IntegerAttr(step));
       state.addAttribute(
           getStyleAttrName(state.name),
           EnvelopeStyleAttr::get(parser.getContext(), style.value()));
       state.addTypes(filteredType);
   
       return success();
   }

   LogicalResult EnvelopeOp::verify()
   {
       int64_t windowSize = getWindowSize();
       int64_t step = getStep();
   
       if (windowSize <= 0)
           return emitOpError("window_size must be positive, got ") << windowSize;
       if (step <= 0) return emitOpError("step must be positive, got ") << step;
   
       int64_t inputSize = getInput().getType().getDimSize(0);
       int64_t filteredSize = getFiltered().getType().getDimSize(0);
   
       if (windowSize > inputSize)
           return emitOpError("window_size (")
                  << windowSize << ") exceeds input size (" << inputSize << ")";
   
       int64_t expectedSize = (inputSize - windowSize) / step + 1;
       if (filteredSize != expectedSize)
           return emitOpError("output size (")
                  << filteredSize << ") does not match expected (" << expectedSize
                  << ") for input_size=" << inputSize
                  << ", window_size=" << windowSize << ", step=" << step;
   
       return success();
   }
