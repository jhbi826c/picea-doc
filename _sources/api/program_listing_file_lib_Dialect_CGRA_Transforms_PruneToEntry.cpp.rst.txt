   //===- PruneToEntry.cpp - keep one outlined family, drop the rest ---------===//
   //
   // A translation unit routinely carries more than the program being built: a
   // hand-written CPU reference (`gemver_cpu`), array initialisers, dead helpers.
   // --annotate-to-cgra wraps offloadable loops wherever it finds them, including
   // inside those, and --outline-cgra-deploy then outlines them too. The result
   // is two complete families of outlined functions, of which exactly one is
   // wanted.
   //
   // Leaving the other in place is not cosmetic. The driver cannot synthesise a
   // call to a reference function it knows nothing about, and — worse, because it
   // is silent — the DFG export emits a .dot for the reference family's loops
   // too, so the mapper numbers bitstreams over a larger set than the driver
   // numbers configurations, and every index past the first extra one pairs the
   // wrong bitstream with the wrong configuration.
   //
   // This used to be handled by spelling the survivors out:
   // `--symbol-privatize=exclude=main,main_init,...,main_stage1,...,main_stage1024`
   // — a thousand-entry list, generated by the Makefile, because symbol-privatize
   // does not glob. The information was in the IR all along: --outline-cgra-deploy
   // stamps `cgra.outlined_from` on every segment it creates. One entry name is
   // therefore enough.
   //
   // Pairs with the upstream `symbol-dce`, which does the actual deletion; this
   // pass only sets visibility.
   //
   //===----------------------------------------------------------------------===//
   
   #include "mlir/Dialect/Func/IR/FuncOps.h"
   #include "mlir/IR/BuiltinOps.h"
   #include "mlir/IR/SymbolTable.h"
   #include "mlir/Pass/Pass.h"
   #include "picea-mlir/Dialect/CGRA/Transforms/Passes.h"
   
   using namespace mlir;
   
   namespace {
   
   class CGRAPruneToEntryPass
           : public PassWrapper<CGRAPruneToEntryPass, OperationPass<ModuleOp>> {
   public:
       MLIR_DEFINE_EXPLICIT_INTERNAL_INLINE_TYPE_ID(CGRAPruneToEntryPass)
   
       CGRAPruneToEntryPass() = default;
       CGRAPruneToEntryPass(StringRef e) { entry = e.str(); }
       CGRAPruneToEntryPass(const CGRAPruneToEntryPass &o) : PassWrapper(o)
       { entry = o.entry; }
   
       StringRef getArgument() const final { return "prune-to-entry"; }
       StringRef getDescription() const final
       {
           return "Privatize every symbol outside the entry function's outlined "
                  "family, so symbol-dce drops it";
       }
   
       Option<std::string> entry{
           *this,
           "entry",
           llvm::cl::desc(
               "Host function the program is built around; it and everything "
               "--outline-cgra-deploy derived from it are kept public"),
           llvm::cl::init("")};
   
       void runOnOperation() override
       {
           // Opt-in: with no entry named there is nothing to prune TO, and
           // silently privatizing the whole module would be far worse than doing
           // nothing.
           if (entry.empty()) return;
           ModuleOp module = getOperation();
           StringRef want = entry;
   
           // The entry function itself need NOT still exist: with
           // --outline-cgra-deploy=drop-host it is erased once its segments are
           // outlined, and only the `cgra.outlined_from` stamps remain. So the
           // sanity check is that SOMETHING claims the name — otherwise a typo
           // would privatise the whole module and symbol-dce would empty it.
           bool found = false;
           for (Operation &op : module.getBody()->getOperations()) {
               auto sym = dyn_cast<SymbolOpInterface>(&op);
               if (!sym) continue;
               if (sym.getName() == want) found = true;
               if (auto from = op.getAttrOfType<StringAttr>("cgra.outlined_from"))
                   if (from.getValue() == want) found = true;
           }
           if (!found) {
               module.emitError("prune-to-entry: nothing in the module is @")
                   << want << " or was outlined from it";
               return signalPassFailure();
           }
   
           // Only visibility is touched here. Deletion is symbol-dce's job, and
           // keeping the two separate means a symbol that IS still referenced
           // survives rather than being erased out from under its user.
           for (Operation &op : module.getBody()->getOperations()) {
               auto sym = dyn_cast<SymbolOpInterface>(&op);
               if (!sym) continue;
               if (sym.getName() == want) continue;
               if (auto from = op.getAttrOfType<StringAttr>("cgra.outlined_from"))
                   if (from.getValue() == want) continue;
               sym.setPrivate();
           }
       }
   };
   
   } // namespace
   
   std::unique_ptr<mlir::Pass> cgra::createCGRAPruneToEntryPass()
   { return std::make_unique<CGRAPruneToEntryPass>(); }
   
   std::unique_ptr<mlir::Pass> cgra::createCGRAPruneToEntryPass(StringRef entry)
   { return std::make_unique<CGRAPruneToEntryPass>(entry); }
