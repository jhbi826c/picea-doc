   void registerCGRAToDriverTranslation()
   {
       TranslateFromMLIRRegistration driver(
           "cgra-to-driver",
           "generate the GR-HEEP STRELA C driver for an outlined CGRA kernel",
           [](Operation* op, raw_ostream &output) {
               return cgra::translateCGRAToDriver(op, output);
           },
           registerCGRADialects);
   
       TranslateFromMLIRRegistration dataset(
           "cgra-to-dataset",
           "generate the dataset.h accompanying the CGRA C driver",
           [](Operation* op, raw_ostream &output) {
               return cgra::translateCGRAToDataset(op, output);
           },
           registerCGRADialects);
   }
   
   } // namespace mlir
