   // The region receives the stream operands as block arguments, so the two must
   // agree in number and type — that is what makes the kernel's data flow
   // explicit.
   LogicalResult cgra::DeployOp::verify()
   {
       Block &body = getBody().front();
       if (body.getNumArguments() != getStreams().size())
           return emitOpError(
                      "region must have one block argument per stream operand (")
                  << body.getNumArguments() << " vs " << getStreams().size()
                  << ")";
       for (auto [arg, stream] : llvm::zip(body.getArguments(), getStreams()))
           if (arg.getType() != stream.getType())
               return emitOpError(
                   "region argument type must match its stream operand");
       return success();
   }
