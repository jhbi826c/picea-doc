   using namespace mlir;

   namespace mlir::base2 {
   namespace {
   
   struct ConstantOpInterface
           : public BufferizableOpInterface::
                 ExternalModel<ConstantOpInterface, ConstantOp> {
       LogicalResult bufferize(
           Operation* op,
           RewriterBase &rewriter,
           const BufferizationOptions &,
           BufferizationState &) const
       {
           auto constantOp = cast<ConstantOp>(op);
           auto type = dyn_cast<RankedTensorType>(constantOp.getType());
           if (!type) return failure();
   
           auto denseBits = dyn_cast<DenseBitsAttr>(constantOp.getValue());
           if (!denseBits)
               return constantOp->emitError(
                   "only dense_bits constants can be bufferized");
   
           auto memrefType = cast<MemRefType>(
               bufferization::getMemRefTypeWithStaticIdentityLayout(type));
           auto newValue = DenseBitsAttr::get(memrefType, denseBits.getData());
           replaceOpWithNewBufferizedOp<ConstantOp>(rewriter, op, newValue);
   
           return success();
       }
   
       bool isWritable(Operation*, Value, const AnalysisState &) const
       { return false; }
   };
   
   } // namespace
   } // namespace mlir::base2
   
   void mlir::base2::registerBufferizableOpInterfaceExternalModels(
       DialectRegistry &registry)
   {
       registry.addExtension(+[](MLIRContext* ctx, base2::Base2Dialect*) {
           ConstantOp::attachInterface<ConstantOpInterface>(*ctx);
       });
   }
