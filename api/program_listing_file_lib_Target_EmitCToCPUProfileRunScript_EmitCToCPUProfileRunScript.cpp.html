// Implementation of the EmitC-to-CPU-profile run-script translation.
//
// @file
// @author     Robert Khasanov (robert.khasanov@tu-dresden.de)

#include "picea-mlir/Target/EmitCToCPUProfileRunScript/CPUProfileRunScriptEmitter.h"

#include "llvm/Support/raw_ostream.h"

using namespace mlir;

LogicalResult
picea::translateEmitCToCPUProfileRunScript(ModuleOp, raw_ostream &os)
{
    os << R"sh(#!/bin/sh

# Generated by resin -t emitc-to-cpu-profile-run-script.
# Builds and runs the generated per-node CPU benchmark.

set -eu

cd "$(dirname "$0")"

processor_type=CortexA53
warmup=3
repetitions=10
samples=5
node=
cpu=

usage()
{
    echo "Usage: ./run.sh [options]"
    echo
    echo "Options:"
    echo "  --processor-type NAME  Mocasin processor type (default: CortexA53)"
    echo "  --warmup N             Warm-up calls per node (default: 3)"
    echo "  --repetitions N        Calls per measurement (default: 10)"
    echo "  --samples N            Measurements per node (default: 5)"
    echo "  --node NAME            Measure only one node"
    echo "  --cpu ID               Pin the benchmark to one logical CPU"
    echo "  -h, --help             Show this help"
}

take_value()
{
    if [ "$#" -lt 2 ]; then
        echo "Error: $1 requires a value" >&2
        exit 1
    fi
}

while [ "$#" -gt 0 ]; do
    case "$1" in
        --processor-type)
            take_value "$@"
            processor_type=$2
            shift 2
            ;;
        --warmup)
            take_value "$@"
            warmup=$2
            shift 2
            ;;
        --repetitions)
            take_value "$@"
            repetitions=$2
            shift 2
            ;;
        --samples)
            take_value "$@"
            samples=$2
            shift 2
            ;;
        --node)
            take_value "$@"
            node=$2
            shift 2
            ;;
        --cpu)
            take_value "$@"
            cpu=$2
            shift 2
            ;;
        -h|--help)
            usage
            exit 0
            ;;
        *)
            echo "Error: unknown option: $1" >&2
            usage >&2
            exit 1
            ;;
    esac
done

case "$processor_type" in
    *[!A-Za-z0-9_.-]*)
        echo "Error: processor type contains characters unsafe for a filename" >&2
        exit 1
        ;;
esac

echo "Building benchmark..."
g++ -O3 -std=c++17 -c nodes.cpp -o nodes.o
g++ -O3 -std=c++17 -c benchmark.cpp -o benchmark.o
g++ nodes.o benchmark.o -o benchmark

set -- --processor-type "$processor_type" \
    --warmup "$warmup" \
    --repetitions "$repetitions" \
    --samples "$samples"
if [ -n "$node" ]; then
    set -- "$@" --node "$node"
fi

profile_file="profiles_${processor_type}.yaml"
echo "Running benchmark with:"
echo "  processor type: $processor_type"
echo "  warmup:         $warmup"
echo "  repetitions:    $repetitions"
echo "  samples:        $samples"
echo "  node:           ${node:-all}"
echo "  output:         $profile_file"

if [ -n "$cpu" ]; then
    if ! command -v taskset >/dev/null 2>&1; then
        echo "Error: --cpu requires taskset" >&2
        exit 1
    fi
    echo "  CPU affinity:   $cpu"
    taskset -c "$cpu" ./benchmark "$@" > "$profile_file"
else
    ./benchmark "$@" > "$profile_file"
fi

cat "$profile_file"
)sh";
    return success();
}
