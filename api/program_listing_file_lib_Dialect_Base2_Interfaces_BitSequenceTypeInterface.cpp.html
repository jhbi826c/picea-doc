namespace {

template<class Float, std::size_t Bits>
struct FloatModel
        : BitSequenceType::ExternalModel<FloatModel<Float, Bits>, Float> {
    static std::size_t getBitWidth(Type) { return Bits; }
};

struct IntModel : BitSequenceType::ExternalModel<IntModel, IntegerType> {
    static std::size_t getBitWidth(Type type)
    {
        const auto intTy = cast<IntegerType>(type);
        return intTy.getWidth();
    }
};

} // namespace

void mlir::base2::registerBitSequenceTypeModels(MLIRContext &ctx)
{
    BFloat16Type::attachInterface<FloatModel<BFloat16Type, 16>>(ctx);
    Float16Type::attachInterface<FloatModel<Float16Type, 16>>(ctx);
    Float32Type::attachInterface<FloatModel<Float32Type, 32>>(ctx);
    Float64Type::attachInterface<FloatModel<Float64Type, 64>>(ctx);
    Float80Type::attachInterface<FloatModel<Float80Type, 80>>(ctx);
    Float128Type::attachInterface<FloatModel<Float128Type, 128>>(ctx);

    IntegerType::attachInterface<IntModel>(ctx);
}
